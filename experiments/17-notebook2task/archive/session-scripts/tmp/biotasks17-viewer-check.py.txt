import json,os,sys
from pathlib import Path
from playwright.sync_api import sync_playwright
root=Path('experiments/17-notebook2task/explorer')
with sync_playwright() as p:
 b=p.chromium.launch(executable_path='/home/exedev/.local/bin/chromium',headless=True,args=['--no-sandbox','--disable-gpu','--no-zygote','--disable-gpu-compositing','--disable-software-rasterizer','--use-gl=disabled','--renderer-process-limit=1'])
 try:
  page=b.new_page(viewport={'width':1280,'height':900});errors=[]
  page.on('pageerror',lambda e:errors.append(str(e)))
  page.on('requestfailed',lambda r:print('REQUEST FAILED',r.url,r.failure,flush=True))
  page.on('console',lambda m:print('CONSOLE',m.type,m.text[:300],flush=True) if m.type=='error' else None)
  page.goto(sys.argv[1] if len(sys.argv)>1 else (root/'index.html').resolve().as_uri(),wait_until='domcontentloaded')
  page.wait_for_selector('#list .card')
  page.select_option('#group','AlphaGenome')
  page.locator('#list .card').filter(has_text='TAL1').click()
  page.click('#toggle-source')
  frame=page.frame_locator('#frame-slot iframe')
  page.wait_for_timeout(8000)
  print([(f.url, f.locator('body').inner_text()[:250]) for f in page.frames],flush=True)
  frame.locator('body').wait_for(timeout=5000)
  assert frame.locator('pre').count() > 10
  assert frame.locator('body').inner_text().count('TAL1')>3
  assert page.locator('#proposal-detail').is_hidden()
  page.locator('#frame-slot').scroll_into_view_if_needed()
  print('BODY STYLE',frame.locator('body').evaluate('(e)=>({display:getComputedStyle(e).display,visibility:getComputedStyle(e).visibility})'))
  page.screenshot(path='/tmp/biotasks17-notebook.png')
  page.click('#toggle-source');assert page.locator('#proposal-detail').is_visible()
  assert not errors, errors
  print(json.dumps({'status':'passed','embedded_source':'TAL1 full nbviewer notebook','return_to_proposals':True,'page_errors':errors}))
 finally:b.close()
